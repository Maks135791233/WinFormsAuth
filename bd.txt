Создать родительскую таблицу:
create table product
(
    id_product serial not null primary key,
    name_product varchar(50) not null,
	cost_product decimal (12,2) not null
);

Вывести данные (создать запрос, который позволит вычислить полную стоимость заказа покупателя с учетом следующей информации:
Количество продукции в заказе
стоимость всех материалов, использованных для производства данной продукции (учитывая норму расхода)
select 
	name_zakazchick,
	zakaz.nomer,
	sum(sostav_zakaza.kol_vo),
	itog,
	string_agg(name_product, ', ')
from zakazchick
inner join zakaz on id_zakazchick = zakazchick_id
inner join sostav_zakaza on id_zakaz = zakaz_id
inner join product on id_product = product_id

group by
	name_zakazchick, zakaz.nomer, itog;
